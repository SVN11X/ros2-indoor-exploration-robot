Mapeo real con robot, primero se debe ejecutar los comandos de mapeo_robot.txt

1° ventana
cd tesis_black_robot
colcon build --symlink-install --executor sequential
source install/setup.bash

2° ventana
source install/setup.bash
ros2 launch articubot_one joystick.launch.py

3° ventana
ros2 run twist_mux twist_mux --ros-args --params-file ./src/articubot_one/config/twist_mux.yaml -r cmd_vel_out:=diff_cont/cmd_vel_unstamped

4° ventana
ros2 launch slam_toolbox online_async_launch.py params_file:=./src/articubot_one/config/mapper_params_online_async.yaml use_sim_time:=false

5° ventana
ros2 launch nav2_bringup navigation_launch.py use_sim_time:=false map_subscribe_transient_local:=true params_file:=/home/dev_ws/tesis_black_robot/src/articubot_one/config/nav2_params.yaml

6° ventana
ros2 run rviz2 rviz2 -d /home/dev_ws/tesis_black_robot/src/articubot_one/config/main.rviz

7° ventana (se ocupa cuando los anteriores esten listos)
source install/setup.bash
ros2 launch articubot_one explore.launch.py use_sim_time:=false
